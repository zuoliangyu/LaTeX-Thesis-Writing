% ===== 字体回退链 =====
% 缺字体时自动回落，避免编译失败。回退路径：
%   宋体类 SimSun → Noto Serif CJK SC（Linux 思源）→ Songti SC（macOS）
%   黑体类 SimHei → Noto Sans CJK SC → Heiti SC
%   仿宋   FangSong → STFangsong（macOS）→ SimSun
%   华文中宋 STZhongsong → SimSun
% 实际使用了哪个字体，编译时会在 .log 里以 \PackageInfo 形式记录。
% 想强制指定字体，把对应 \IfFontExistsTF 块替换为 \setCJKmainfont{...} 即可。

% 预探测：把候选字体是否存在做成布尔开关。这一步不能放在 \newcommand 内部，
% 因为 fontspec 的 \IfFontExistsTF 在 false 分支会触发 expl3 clist 映射，
% 与外层 \newcommand 的参数 # 系统冲突 ("Illegal parameter number")。
\makeatletter
\newif\ifthesis@hasSimSun        \IfFontExistsTF{SimSun}{\thesis@hasSimSuntrue}{}
\newif\ifthesis@hasSimHei        \IfFontExistsTF{SimHei}{\thesis@hasSimHeitrue}{}
\newif\ifthesis@hasFangSong      \IfFontExistsTF{FangSong}{\thesis@hasFangSongtrue}{}
\newif\ifthesis@hasSTZhongsong   \IfFontExistsTF{STZhongsong}{\thesis@hasSTZhongsongtrue}{}
\newif\ifthesis@hasSTSong        \IfFontExistsTF{STSong}{\thesis@hasSTSongtrue}{}
\newif\ifthesis@hasSTFangsong    \IfFontExistsTF{STFangsong}{\thesis@hasSTFangsongtrue}{}
\newif\ifthesis@hasNotoSerifCJK  \IfFontExistsTF{Noto Serif CJK SC}{\thesis@hasNotoSerifCJKtrue}{}
\newif\ifthesis@hasNotoSansCJK   \IfFontExistsTF{Noto Sans CJK SC}{\thesis@hasNotoSansCJKtrue}{}
\newif\ifthesis@hasSongtiSC      \IfFontExistsTF{Songti SC}{\thesis@hasSongtiSCtrue}{}
\newif\ifthesis@hasHeitiSC       \IfFontExistsTF{Heiti SC}{\thesis@hasHeitiSCtrue}{}
\newif\ifthesis@hasTimes         \IfFontExistsTF{Times New Roman}{\thesis@hasTimestrue}{}
\newif\ifthesis@hasArial         \IfFontExistsTF{Arial}{\thesis@hasArialtrue}{}
\newif\ifthesis@hasCourier       \IfFontExistsTF{Courier New}{\thesis@hasCouriertrue}{}
\newif\ifthesis@hasLibSerif      \IfFontExistsTF{Liberation Serif}{\thesis@hasLibSeriftrue}{}
\newif\ifthesis@hasLibSans       \IfFontExistsTF{Liberation Sans}{\thesis@hasLibSanstrue}{}
\newif\ifthesis@hasLibMono       \IfFontExistsTF{Liberation Mono}{\thesis@hasLibMonotrue}{}
\newif\ifthesis@hasGyreTermes    \IfFontExistsTF{TeX Gyre Termes}{\thesis@hasGyreTermestrue}{}
\newif\ifthesis@hasGyreHeros     \IfFontExistsTF{TeX Gyre Heros}{\thesis@hasGyreHerostrue}{}
\newif\ifthesis@hasGyreCursor    \IfFontExistsTF{TeX Gyre Cursor}{\thesis@hasGyreCursortrue}{}

% ----- 中文主字体（宋体）：SimSun → 思源宋体 → Songti SC -----
\ifthesis@hasSimSun
  \setCJKmainfont{SimSun}[AutoFakeBold=2.5]
\else\ifthesis@hasNotoSerifCJK
  \setCJKmainfont{Noto Serif CJK SC}[AutoFakeBold=2.5]
  \PackageWarning{thesis-fonts}{SimSun missing, fallback to Noto Serif CJK SC}
\else\ifthesis@hasSongtiSC
  \setCJKmainfont{Songti SC}[AutoFakeBold=2.5]
  \PackageWarning{thesis-fonts}{SimSun missing, fallback to Songti SC}
\else
  \PackageError{thesis-fonts}{No CJK serif font found (SimSun / Noto Serif CJK SC / Songti SC)}%
    {Install one, or edit setup/format.tex.}
\fi\fi\fi

% ----- 中文无衬线（黑体）：SimHei → 思源黑体 → Heiti SC -----
\ifthesis@hasSimHei
  \setCJKsansfont{SimHei}[AutoFakeBold=2.5]
\else\ifthesis@hasNotoSansCJK
  \setCJKsansfont{Noto Sans CJK SC}[AutoFakeBold=2.5]
  \PackageWarning{thesis-fonts}{SimHei missing, fallback to Noto Sans CJK SC}
\else\ifthesis@hasHeitiSC
  \setCJKsansfont{Heiti SC}[AutoFakeBold=2.5]
  \PackageWarning{thesis-fonts}{SimHei missing, fallback to Heiti SC}
\else
  \PackageError{thesis-fonts}{No CJK sans font found}{Install SimHei / Noto Sans CJK SC.}
\fi\fi\fi

% ----- 中文等宽（仿宋）：FangSong → STFangsong → SimSun -----
\ifthesis@hasFangSong
  \setCJKmonofont{FangSong}
\else\ifthesis@hasSTFangsong
  \setCJKmonofont{STFangsong}
  \PackageWarning{thesis-fonts}{FangSong missing, fallback to STFangsong}
\else\ifthesis@hasSimSun
  \setCJKmonofont{SimSun}
  \PackageWarning{thesis-fonts}{FangSong / STFangsong missing, fallback to SimSun}
\else
  \PackageError{thesis-fonts}{No CJK mono font found}{Install FangSong or SimSun.}
\fi\fi\fi

% ----- \heiti -----
\ifthesis@hasSimHei
  \newCJKfontfamily{\heiti}{SimHei}[AutoFakeBold=2.5]
\else\ifthesis@hasNotoSansCJK
  \newCJKfontfamily{\heiti}{Noto Sans CJK SC}[AutoFakeBold=2.5]
\else\ifthesis@hasHeitiSC
  \newCJKfontfamily{\heiti}{Heiti SC}[AutoFakeBold=2.5]
\else
  \PackageError{thesis-fonts}{No font available for \string\heiti}{Install SimHei.}
\fi\fi\fi

% ----- \songti -----
\ifthesis@hasSimSun
  \newCJKfontfamily{\songti}{SimSun}
\else\ifthesis@hasNotoSerifCJK
  \newCJKfontfamily{\songti}{Noto Serif CJK SC}
\else\ifthesis@hasSongtiSC
  \newCJKfontfamily{\songti}{Songti SC}
\else
  \PackageError{thesis-fonts}{No font available for \string\songti}{Install SimSun.}
\fi\fi\fi

% ----- \fangsong -----
\ifthesis@hasFangSong
  \newCJKfontfamily{\fangsong}{FangSong}
\else\ifthesis@hasSTFangsong
  \newCJKfontfamily{\fangsong}{STFangsong}
\else\ifthesis@hasSimSun
  \newCJKfontfamily{\fangsong}{SimSun}
  \PackageWarning{thesis-fonts}{FangSong missing for \string\fangsong, using SimSun}
\else
  \PackageError{thesis-fonts}{No font available for \string\fangsong}{Install FangSong.}
\fi\fi\fi

% ----- \huawenzhongsong（封面校名常用）：STZhongsong → STSong → SimSun -----
\ifthesis@hasSTZhongsong
  \newCJKfontfamily{\huawenzhongsong}{STZhongsong}
\else\ifthesis@hasSTSong
  \newCJKfontfamily{\huawenzhongsong}{STSong}
  \PackageWarning{thesis-fonts}{STZhongsong missing, fallback to STSong (cover may look slightly different)}
\else\ifthesis@hasSimSun
  \newCJKfontfamily{\huawenzhongsong}{SimSun}
  \PackageWarning{thesis-fonts}{STZhongsong missing, fallback to SimSun (cover may look slightly different)}
\else
  \PackageError{thesis-fonts}{No font available for \string\huawenzhongsong}{Install STZhongsong or SimSun.}
\fi\fi\fi

% ----- 西文：Times / Liberation Serif / TeX Gyre Termes -----
\ifthesis@hasTimes
  \setmainfont{Times New Roman}
\else\ifthesis@hasLibSerif
  \setmainfont{Liberation Serif}
  \PackageWarning{thesis-fonts}{Times New Roman missing, fallback to Liberation Serif}
\else\ifthesis@hasGyreTermes
  \setmainfont{TeX Gyre Termes}
  \PackageWarning{thesis-fonts}{Times New Roman missing, fallback to TeX Gyre Termes}
\else
  \PackageError{thesis-fonts}{No serif Latin font found}{Install Times New Roman.}
\fi\fi\fi

% ----- 西文 sans -----
\ifthesis@hasArial
  \setsansfont{Arial}
\else\ifthesis@hasLibSans
  \setsansfont{Liberation Sans}
  \PackageWarning{thesis-fonts}{Arial missing, fallback to Liberation Sans}
\else\ifthesis@hasGyreHeros
  \setsansfont{TeX Gyre Heros}
  \PackageWarning{thesis-fonts}{Arial missing, fallback to TeX Gyre Heros}
\else
  \PackageError{thesis-fonts}{No sans Latin font found}{Install Arial.}
\fi\fi\fi

% ----- 西文 mono -----
\ifthesis@hasCourier
  \setmonofont{Courier New}
\else\ifthesis@hasLibMono
  \setmonofont{Liberation Mono}
  \PackageWarning{thesis-fonts}{Courier New missing, fallback to Liberation Mono}
\else\ifthesis@hasGyreCursor
  \setmonofont{TeX Gyre Cursor}
  \PackageWarning{thesis-fonts}{Courier New missing, fallback to TeX Gyre Cursor}
\else
  \PackageError{thesis-fonts}{No mono Latin font found}{Install Courier New.}
\fi\fi\fi
\makeatother
