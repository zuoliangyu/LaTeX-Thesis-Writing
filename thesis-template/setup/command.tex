% ---------- TikZ 绘图样式预设 ----------

\tikzset{
    flow/start/.style   = {rectangle, rounded corners, draw=black, thick,
                            minimum width=2.4cm, minimum height=0.8cm,
                            align=center, font=\small},
    flow/proc/.style    = {rectangle, draw=black, thick,
                            minimum width=2.6cm, minimum height=0.8cm,
                            align=center, font=\small},
    flow/decide/.style  = {diamond, draw=black, thick, aspect=2,
                            inner sep=1pt, align=center, font=\small},
    flow/io/.style      = {trapezium, trapezium left angle=70, trapezium right angle=110,
                            draw=black, thick, minimum width=2.4cm, minimum height=0.8cm,
                            align=center, font=\small},
    flow/arrow/.style   = {-{Latex[length=2mm]}, thick},
    % 系统框图节点：模块、总线、信号
    block/module/.style = {rectangle, draw=black, thick, fill=blue!8,
